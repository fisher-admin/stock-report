```bash
import pandas as pd, akshare as ak, glob
B="stock_data/03-working/backtest_cache/"; days="$DAYS".split()

have={g.split('daily_')[1][:8] for g in glob.glob(f"{B}daily_*.parquet")}
for d in days:
    prev=tdays[tdays.index(d)-1]
    if prev not in have: print(d,"prev",prev,"NO_FILE"); continue
    cur=pd.read_parquet(f"{B}daily_{d}.parquet").set_index('ts_code')
    pv=pd.read_parquet(f"{B}daily_{prev}.parquet").set_index('ts_code')
    c=cur.index.intersection(pv.index)[:300]
    print(f"{d} vs {prev}: mism={int((cur.loc[c,'pre_close']-pv.loc[c,'close']).abs().gt(0.01).sum())}/{len(c)}")
PY
```
